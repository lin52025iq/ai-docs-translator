import { Callout } from '@/components/ui/Callout'

了解如何构建一个侧边栏组件，列出文档中所有待处理的建议，显示每项更改的作者，并允许用户接受或拒绝建议。

<Callout title="框架说明" variant="hint">
  本指南使用 React，但同样的概念也适用于 Vue 和 Tiptap 支持的其他框架。
</Callout>

## 前提条件

本指南假设你已经有一个可正常使用、且启用了 Tracked Changes 扩展的编辑器。如果还没有，请先参阅[编辑器设置指南](/tracked-changes/guides/editor-setup)。

## 查询建议

该扩展导出了 `findSuggestions` 工具，可返回文档中的所有建议。在 `onCreate` 和 `onUpdate` 中调用它，即可让列表保持同步。

```jsx
import { useState } from 'react'
import { useEditor, EditorContent } from '@tiptap/react'
import StarterKit from '@tiptap/starter-kit'
import { TrackedChanges, findSuggestions } from '@tiptap-pro/extension-tracked-changes'

function Editor() {
  const [suggestions, setSuggestions] = useState([])

  const editor = useEditor({
    extensions: [
      StarterKit,
      TrackedChanges.configure({
        enabled: true,
        userId: 'user-123',
        userMetadata: { name: 'John Doe' },
      }),
    ],
    content: '<p>Start editing to see tracked changes.</p>',
    onCreate: ({ editor: currentEditor }) => {
      setSuggestions(findSuggestions(currentEditor))
    },
    onUpdate: ({ editor: currentEditor }) => {
      setSuggestions(findSuggestions(currentEditor))
    },
  })

  // Deduplicate — a suggestion can span multiple text nodes
  const uniqueSuggestions = Array.from(new Map(suggestions.map((s) => [s.id, s])).values())

  if (!editor) {
    return null
  }

  return (
    <div className="editor-layout">
      <div className="editor-content">
        <EditorContent editor={editor} />
      </div>
      <SuggestionList
        suggestions={uniqueSuggestions}
        onAccept={(id) => {
          editor.commands.acceptSuggestion({ id })
          setSuggestions(findSuggestions(editor))
        }}
        onReject={(id) => {
          editor.commands.rejectSuggestion({ id })
          setSuggestions(findSuggestions(editor))
        }}
      />
    </div>
  )
}
```

`★ Insight ─────────────────────────────────────`
当一项建议跨越多个文本节点时（例如部分文本为粗体），它可能涉及多个文本节点。`findSuggestions` 会为每个文本节点返回一条记录，因此必须按 `id` 去重，避免列表中重复显示同一项建议。
`─────────────────────────────────────────────────`

## 构建 SuggestionList 组件

每项建议都包含 `type`、发生更改的 `text`、用户元数据和时间戳。属性更改建议还会提供发生更改的节点类型和属性值。你可以利用这些信息来呈现清晰的列表。

```jsx
function SuggestionList({ suggestions, onAccept, onReject }) {
  if (suggestions.length === 0) {
    return (
      <div className="suggestion-list">
        <p>No pending suggestions</p>
      </div>
    )
  }

  return (
    <div className="suggestion-list">
      <h3>Suggestions ({suggestions.length})</h3>
      {suggestions.map((suggestion) => (
        <SuggestionItem
          key={suggestion.id}
          suggestion={suggestion}
          onAccept={() => onAccept(suggestion.id)}
          onReject={() => onReject(suggestion.id)}
        />
      ))}
    </div>
  )
}
```

## 构建 SuggestionItem 组件

显示建议类型、文本预览、用户信息和操作按钮。

当建议影响块级节点（例如段落或标题）时，建议会提供 `insertedNodes` 或 `deletedNodes`，它们是用于描述完整节点结构的 `JSONContent` 数组。你可以使用这些数据，在内容文本上方显示节点类型徽标。

```jsx
function SuggestionItem({ suggestion, onAccept, onReject }) {
  const typeLabels = {
    add: '+ Insertion',
    delete: '− Deletion',
    replace: '~ Replacement',
    markChange: '~ Formatting change',
    attributeChange: '~ Attribute change',
    sink: '→ List indent',
    lift: '← List outdent',
  }

  const typeLabel =
    suggestion.type === 'split'
      ? `~ ${suggestion.splitKind === 'insertion' ? 'new' : 'split'} ${suggestion.nodeType ?? 'node'}`
      : typeLabels[suggestion.type]

  // Derive a node type label from block-level node suggestions
  const nodes = suggestion.insertedNodes ?? suggestion.deletedNodes
  const nodeType = suggestion.nodeType ?? nodes?.[0]?.type

  return (
    <div className="suggestion-item">
      <div className="suggestion-item__type">{typeLabel}</div>

      {nodeType && <span className="suggestion-item__node-badge">{nodeType}</span>}

      <div className="suggestion-item__text">
        {suggestion.type === 'attributeChange' ? (
          <AttributeChanges changes={suggestion.attributeChanges ?? []} />
        ) : (
          <>
            "{suggestion.text}"
            {suggestion.type === 'replace' && suggestion.replacedText && (
              <span className="suggestion-item__replaced-text">
                {' '}
                (was: "{suggestion.replacedText}")
              </span>
            )}
          </>
        )}
      </div>

      <div className="suggestion-item__meta">
        {suggestion.userMetadata?.name || suggestion.userId}
        {' · '}
        {new Date(suggestion.createdAt).toLocaleTimeString()}
      </div>

      <div className="suggestion-item__actions">
        <button onClick={onAccept}>Accept</button>
        <button onClick={onReject}>Reject</button>
      </div>
    </div>
  )
}

function AttributeChanges({ changes }) {
  return changes.map((change) => (
    <div key={change.attributeName}>
      {change.attributeName}: {JSON.stringify(change.oldValue)} → {JSON.stringify(change.newValue)}
    </div>
  ))
}
```

使用 `suggestion.text` 可以单独显示每位作者所做的贡献，这是审阅面板的合适默认选项。如果另一位作者在这项建议中又叠加了编辑，`suggestion.fullText` 就会与 `suggestion.text` 不同，并包含嵌套文本在内的完整范围。你可以将其作为提示显示：

```jsx
{
  suggestion.fullText !== suggestion.text && (
    <span className="suggestion-item__stacked-hint">
      with stacked edits: "{suggestion.fullText}"
    </span>
  )
}
```

完整行为请参阅 [`text` 与 `fullText`](/tracked-changes/api-reference/types#text-vs-fulltext) 和[嵌套建议](/tracked-changes/usage/advanced-usage#nested-and-stacked-suggestions)。

## 添加批量操作

在列表上方添加按钮，以便一次性接受或拒绝所有建议。

```jsx
function SuggestionList({ suggestions, onAccept, onReject, onAcceptAll, onRejectAll }) {
  return (
    <div className="suggestion-list">
      <h3>Suggestions ({suggestions.length})</h3>

      {suggestions.length > 0 && (
        <div className="suggestion-list__bulk-actions">
          <button onClick={onAcceptAll}>Accept all</button>
          <button onClick={onRejectAll}>Reject all</button>
        </div>
      )}

      {suggestions.map((suggestion) => (
        <SuggestionItem
          key={suggestion.id}
          suggestion={suggestion}
          onAccept={() => onAccept(suggestion.id)}
          onReject={() => onReject(suggestion.id)}
        />
      ))}
    </div>
  )
}
```

在父组件中接入批量操作：

```jsx
<SuggestionList
  suggestions={uniqueSuggestions}
  onAccept={(id) => {
    editor.commands.acceptSuggestion({ id })
    setSuggestions(findSuggestions(editor))
  }}
  onReject={(id) => {
    editor.commands.rejectSuggestion({ id })
    setSuggestions(findSuggestions(editor))
  }}
  onAcceptAll={() => {
    editor.commands.acceptAllSuggestions()
    setSuggestions([])
  }}
  onRejectAll={() => {
    editor.commands.rejectAllSuggestions()
    setSuggestions([])
  }}
/>
```

## 筛选建议

使用 `findSuggestions` 的 options 参数，按类型或用户筛选：

```jsx
import { findSuggestions } from '@tiptap-pro/extension-tracked-changes'

// Only insertions
const insertions = findSuggestions(editor, 'suggestion', { type: 'add' })

// Only deletions
const deletions = findSuggestions(editor, 'suggestion', { type: 'delete' })

// Only from a specific user
const userSuggestions = findSuggestions(editor, 'suggestion', { userId: 'user-123' })
```

你可以在侧边栏中添加筛选控件，让用户查看特定类型的建议。

## 监听事件

如需更及时地更新，可以监听扩展事件，替代（或补充）在 `onUpdate` 中轮询：

```jsx
import { useEffect } from 'react'

useEffect(() => {
  if (!editor) return

  const updateSuggestions = () => {
    setSuggestions(findSuggestions(editor))
  }

  editor.on('trackedChanges:suggestionCreated', updateSuggestions)
  editor.on('trackedChanges:suggestionAccepted', updateSuggestions)
  editor.on('trackedChanges:suggestionRejected', updateSuggestions)
  editor.on('trackedChanges:suggestionRemoved', updateSuggestions)
  editor.on('trackedChanges:suggestionsUpdated', updateSuggestions)

  return () => {
    editor.off('trackedChanges:suggestionCreated', updateSuggestions)
    editor.off('trackedChanges:suggestionAccepted', updateSuggestions)
    editor.off('trackedChanges:suggestionRejected', updateSuggestions)
    editor.off('trackedChanges:suggestionRemoved', updateSuggestions)
    editor.off('trackedChanges:suggestionsUpdated', updateSuggestions)
  }
}, [editor])
```
