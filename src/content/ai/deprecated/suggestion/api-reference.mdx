## 扩展选项

```ts
/**
 * AI Suggestion 扩展的配置选项。
 */
export interface AiSuggestionOptions {
  /**
   * 校对时应用的规则
   * @default []
   */
  rules: AiSuggestionRule[]
  /**
   * 在编辑器中显示的初始建议
   * @default []
   */
  initialSuggestions: Suggestion[]
  /**
   * 开始校对前应用的初始拒绝项
   * @default []
   */
  initialRejections: AiSuggestionRejection[]
  /**
   * 用于自定义编辑器中建议装饰效果的函数
   * @default `getDefaultDecorations()`
   * @param options - 自定义建议装饰效果的选项。
   * @return `Decoration` 对象数组。
   */
  getCustomSuggestionDecoration: (options: GetCustomSuggestionDecorationOptions) => Decoration[]
  /**
   * 内容更改后，重新加载建议前等待的时间（毫秒）。
   * @default 800
   */
  debounceTimeout: number
  /**
   * 是否在编辑器初始化时加载建议
   * @default true
   */
  loadOnStart: boolean

  /**
   * 编辑器内容更改时是否重新加载建议
   * @default true
   */
  reloadOnUpdate: boolean

  /**
   * 加载建议时处理错误的回调。如果加载建议的操作已中止，
   * 则不会调用此回调。
   *
   * @param error - 加载建议时发生的错误。
   * @param context - 操作上下文，包括 Editor 实例
   * @default console.error
   */
  onLoadSuggestionsError: (error: unknown, context: AiSuggestionEventContext) => void
  /**
   * 用于校对内容并生成建议的 AI 模型。
   * @default "gpt-4o-mini"
   */
  modelName: AiSuggestionModelName
  /**
   * 根据当前编辑器内容和规则，从外部来源加载建议的函数。借助此函数，
   * 你可以使用自己的 AI 模型分析内容并返回建议。
   *
   * @param options - 用于解析建议的选项。
   * @return 应用的建议列表。
   */
  resolver: (options: AiSuggestionCustomResolverOptions) => Promise<Suggestion[]>
  /**
   * Tiptap AI 令牌。
   */
  token: string
  /**
   * Tiptap AI API 的基础 URL。
   */
  baseUrl: string
  /**
   * 可选上下文，为 AI 模型提供生成建议所需的补充信息。
   * @default null
   */
  context: string | null
  /**
   * 是否将编辑器内容分块，并缓存每个块的建议。这样在重新加载内容时，
   * 扩展可以复用各个块缓存的建议。对于大型文档，此功能有助于提升性能并降低成本。
   *
   * @default true
   */
  enableCache: boolean
  /**
   * 文档的分块大小，以编辑器内容的顶层子节点数量为单位。
   *
   * @default 2
   */
  chunkSize: number
  /**
   * 将编辑器的 HTML 内容拆分为多个块的函数。默认情况下，它会根据配置的
   * 分块大小将 HTML 拆分成更小的部分，并返回 HTML 块数组。你可以提供自己的函数
   * 来覆盖此行为。
   *
   * @param options - HTML 分块选项。
   * @returns 分块数组
   * @default `defaultChunkHtmlFunction`
   */
  chunkHtmlFunction: (options: ChunkHtmlOptions) => HtmlChunk[]
}
```

## 扩展命令

````ts
declare module '@tiptap/core' {
  interface Commands<ReturnType> {
    aiSuggestion: {
      /**
       * 立即加载 AI 建议。
       */
      loadAiSuggestions: (options?: LoadAiSuggestionsOptions) => ReturnType
      /**
       * 在扩展选项中定义的防抖等待时间后加载 AI 建议。
       */
      loadAiSuggestionsDebounced: (options?: LoadAiSuggestionsOptions) => ReturnType
      /**
       * 设置要显示的 AI 建议。
       *
       * @param suggestions 要显示的建议。
       */
      setAiSuggestions: (suggestions: Suggestion[]) => ReturnType
      /**
       * 将某条建议设为选中状态。选中的建议在编辑器中会显示为不同样式。
       * */
      selectAiSuggestion: (suggestionId: string) => ReturnType
      /**
       * 将建议应用到编辑器，并修改其内容。
       *
       * @param options 建议 ID 和选中的替换选项。如果未提供替换选项，
       * 则使用第一个选项。还可以自定义替换内容的格式。
       */
      applyAiSuggestion: (options: ApplyAiSuggestionOptions) => ReturnType
      /**
       * 将建议标记为已拒绝，并从建议列表中移除。
       *
       * 此操作会将建议添加到拒绝列表中，以便 AI 再次生成该建议时将其忽略。
       *
       * @param suggestionId 被拒绝建议的 ID
       */
      rejectAiSuggestion: (suggestionId: string) => ReturnType

      /**
       * 删除建议，并将其从建议列表中移除。
       *
       * @param suggestionId 要删除的建议 ID
       */
      removeAiSuggestion: (suggestionId: string) => ReturnType

      /**
       * 设置要拒绝的建议。此命令可用于撤销拒绝操作或清除所有拒绝项。
       *
       * @param rejections 要设置的拒绝项。
       */
      setAiSuggestionRejections: (rejections: AiSuggestionRejection[]) => ReturnType

      /**
       * 应用所有未被拒绝的 AI 建议。每条建议均应用第一个替换选项。
       */
      applyAllAiSuggestions: (options?: ApplyAllAiSuggestionOptions) => ReturnType
      /**
       * 设置 AI Suggestion 规则。无需重新加载编辑器，即可更新用于校对的规则。
       *
       * 此命令不会重新加载建议。若要使用新规则重新加载建议，请调用
       * `loadAiSuggestions` 命令，如下所示：
       *
       * ```js
       * editor.chain().setAiSuggestionRules(newRules).loadAiSuggestions().run()
       * ```
       */
      setAiSuggestionRules: (rules?: AiSuggestionRule[]) => ReturnType

      /**
       * 设置 AI Suggestion 上下文。此上下文用于向 AI 模型提供生成建议所需的补充信息。
       *
       * 此命令不会重新加载建议。若要使用新上下文重新加载建议，请调用
       * `loadAiSuggestions` 命令，如下所示：
       *
       * ```js
       * editor.chain().setAiSuggestionContext(newContext).loadAiSuggestions().run()
       * ```
       *
       * @param context 为 AI 模型提供补充信息的可选上下文
       */
      setAiSuggestionContext: (context: string | null) => ReturnType

      /**
       * 设置编辑器内容更改时是否重新加载建议。
       * 这样可以动态启用或禁用内容更新时自动重新加载建议。
       *
       * @param reloadOnUpdate 内容更新时是否重新加载建议
       * @default true
       */
      setAiSuggestionReloadOnUpdate: (reloadOnUpdate: boolean) => ReturnType
    }
  }
}

export interface ApplyAiSuggestionOptions {
  /**
   * 要应用的建议标识符。如果找不到该建议，此方法将不执行任何操作。
   */
  suggestionId: string
  /**
   * 要应用的替换选项标识符。如果为 `undefined`，
   * 则默认使用第一个替换选项。
   */
  replacementOptionId?: string
  /**
   * 确定建议的应用方式。
   * 如果为 `rich-text`，建议将按 HTML 格式处理。
   * 如果为 `plain-text`，建议将按纯文本格式处理。
   * @default "plain-text"
   */
  format?: 'rich-text' | 'plain-text'
}

export interface ApplyAllAiSuggestionOptions {
  /**
   * 确定建议的应用方式。
   * 如果为 `rich-text`，建议将按 HTML 格式处理。
   * 如果为 `plain-text`，建议将按纯文本格式处理。
   * @default "plain-text"
   */
  format?: 'rich-text' | 'plain-text'
}
````

## 扩展存储

```ts
/**
 * AI Suggestion 扩展的内部存储。
 */
export interface AiSuggestionStorage {
  /**
   * 扩展使用的 ProseMirror 插件键
   */
  pluginKey: PluginKey<AiSuggestionProsemirrorPluginState>
  /**
   * 校对时应用的规则
   */
  rules: AiSuggestionRule[]
  /**
   * 获取当前建议的函数
   */
  getSuggestions: () => Suggestion[]
  /**
   * 获取已拒绝建议的函数
   */
  getRejections: () => AiSuggestionRejection[]
  /**
   * 获取当前选中建议的函数
   */
  getSelectedSuggestion: () => Suggestion | null
  /**
   * 当前是否正在加载建议
   */
  isLoading: boolean
  /**
   * 是否至少加载过一次建议
   */
  firstLoad: boolean
  /**
   * 上次尝试加载时发生的错误（如果有）
   */
  error: unknown | null
  /**
   * 用于加载建议的防抖函数
   */
  debouncedFunction: DebouncedFunction<(options?: LoadAiSuggestionsOptions) => void>
  /**
   * 用于中止加载建议的控制器
   */
  abortController: AbortController
  /**
   * 可选上下文，为 AI 模型提供生成建议所需的补充信息。
   */
  context: string | null
  /**
   * 用于存储文档各个块的建议的缓存。
   */
  cache: AiSuggestionCache
  /**
   * 编辑器内容更改时是否重新加载建议
   */
  reloadOnUpdate: boolean
}

/**
 * 用于在编辑器中创建建议装饰效果的参数。
 */
export interface GetCustomSuggestionDecorationOptions {
  /** 正在装饰的建议 */
  suggestion: Suggestion
  /** 此建议当前是否处于选中状态 */
  isSelected: boolean
  /** 获取扩展提供的默认装饰样式的函数 */
  getDefaultDecorations: () => Decoration[]
}
```

## 扩展类型

### 规则

```ts
/**
 * 校对时应用的规则。
 */
export interface AiSuggestionRule {
  /**
   * 规则的唯一标识符。
   */
  id: string
  /**
   * 规则的标题。不会作为提示词的一部分发送给 LLM 模型，
   * 仅用于显示。
   */
  title: string
  /**
   * 规则的提示词。将发送给 AI 模型用于校对。
   */
  prompt: string
  /**
   * 规则的颜色。用于给应用规则时需要替换的内容添加下划线。
   */
  color: string
  /**
   * 规则的背景色。选中建议时，用于突出显示内容。
   */
  backgroundColor: string
  /**
   * 向 Tiptap Cloud 中的 AI 模型发送请求时是否忽略此规则。
   *
   * 如果为 `true`，则不会将此规则发送给 Tiptap Cloud 中的 AI 模型。
   *
   * @default false
   */
  ignoreInTiptapCloud?: boolean
  /**
   * 以差异格式显示此规则的建议，其中要删除的内容显示为红色，
   * 要添加的内容显示为绿色。
   *
   * @default false
   */
  displayAsDiff?: boolean
  /**
   * 规则的额外元数据，可用于存储其他信息（例如来源或类别）。扩展内部不会使用这些信息，
   * 但开发者可以利用它们自定义规则在 UI 中的显示方式。
   */
  metadata?: any
}
```

### 校对建议

```ts
import { Range } from '@tiptap/core'
import { Slice } from '@tiptap/pm/model'

import { AiSuggestionRule } from './ai-suggestion-rule'

/**
 * 建议的替换选项。包含要添加的文本和要替换的切片。
 */
export interface AiSuggestionReplacementOption {
  id: string
  /**
   * 替换格式为 `plain-text` 时要添加的文本。
   */
  addText: string
  /**
   * 替换格式为 `rich-text` 时要添加的内容。
   */
  addSlice: Slice
}

/**
 * 来自 AI Suggestion 扩展的建议。包含要替换的范围、要删除的文本、替换选项，
 * 以及要应用的规则。
 *
 * 一条建议可以包含多个替换选项。用户可以选择其中一个来应用建议。
 */
export interface Suggestion {
  /**
   * 建议的唯一标识符。
   */
  id: string
  /**
   * 编辑器中应被替换的内容范围。
   */
  deleteRange: Range
  /**
   * 要从编辑器中删除的纯文本内容
   */
  deleteText: string
  /**
   * 要从编辑器中删除的内容，以 ProseMirror 切片形式表示。
   */
  deleteSlice: Slice
  /**
   * 此建议的多个替换选项。用户可以选择其中一个来应用建议。
   */
  replacementOptions: AiSuggestionReplacementOption[]
  /**
   * 应用此建议时遵循的校对规则。
   */
  rule: AiSuggestionRule
  /**
   * 此建议是否已被用户拒绝。
   */
  isRejected: boolean
  /**
   * 建议的额外元数据，可用于存储其他信息（例如来源或类别）。扩展内部不会使用这些信息，
   * 但开发者可以利用它们自定义建议在 UI 中的显示方式。
   */
  metadata?: any
}
```

### 已拒绝的建议

```ts
import { AiSuggestionRule } from './ai-suggestion-rule'

/**
 * 表示一条已拒绝的建议。这些建议会存储在 AI Suggestion 扩展的存储中，
 * 以避免再次生成。
 */
export interface AiSuggestionRejection {
  /**
   * 被拒绝建议所遵循的规则。
   */
  rule: AiSuggestionRule
  /**
   * 被拒绝建议原本要删除的文本。
   */
  deleteText: string
}
```
