## 扩展选项

```ts
/**
 * AI Changes 扩展的配置选项
 */
export interface AiChangesOptions {
  /**
   * 创建 AI 更改的装饰
   *
   * @default getDefaultDecorations 参数
   *
   * @param options 当前更改、更改列表、当前更改是否已选中、用于获取默认装饰的函数、编辑器实例、AI 更改前的文档，以及 AI 更改后的当前文档。
   * @returns 要应用于文档的 ProseMirror 装饰列表
   */
  getCustomDecorations: (options: GetCustomDecorationsOptions) => Decoration[]
}

/**
 * `getCustomDecorations` 函数的选项
 */
export interface GetCustomDecorationsOptions {
  /**
   * 当前正在处理的更改
   */
  change: AiChange
  /**
   * AI 所做的更改列表
   */
  changes: AiChange[]
  /**
   * 当前更改是否已选中
   */
  isSelected: boolean
  /**
   * 获取扩展在未提供 `getCustomDecorations` 函数时会应用于文档的默认装饰。
   *
   * @returns 要应用于文档的默认装饰列表
   */
  getDefaultDecorations: () => Decoration[]
  /**
   * Tiptap 编辑器实例
   */
  editor: Editor
  /**
   * AI 更改前的文档
   */
  previousDoc: Node | null
  /**
   * AI 更改后的当前文档
   */
  currentDoc: Node
}
```

## 扩展命令

```ts
/**
 * AI Changes 扩展的命令
 */
declare module '@tiptap/core' {
  interface Commands<ReturnType> {
    aiChanges: {
      /**
       * 开始跟踪文档中的 AI 更改，并显示编辑器中当前文档与初始文档之间的差异。
       *
       * @param doc 用于比较的初始文档。如果未提供，则使用当前文档。
       */
      startTrackingAiChanges: (doc?: Node) => ReturnType

      /**
       * 停止跟踪文档中的 AI 更改。将初始文档设为 `null`，并停止显示当前文档与初始文档之间的差异。
       */
      stopTrackingAiChanges: () => ReturnType

      /**
       * 接受文档中的所有 AI 更改。将初始文档设为当前文档。
       */
      acceptAllAiChanges: () => ReturnType

      /**
       * 撤销文档中的所有 AI 更改，并恢复初始文档。
       */
      rejectAllAiChanges: () => ReturnType

      /**
       * 接受文档中的特定 AI 更改。修改初始文档，使其在该更改处与当前文档内容一致。
       * @param id 要接受的更改 ID
       */
      acceptAiChange: (id: string) => ReturnType

      /**
       * 拒绝文档中的特定 AI 更改。当前文档在该更改处的内容将与初始文档一致。
       *
       * @param id 要拒绝的更改 ID
       * @returns
       */
      rejectAiChange: (id: string) => ReturnType

      /**
       * 显示或隐藏文档中的 AI 更改。
       *
       * @param show 是否显示 AI 更改
       */
      setShowAiChanges: (show: boolean) => ReturnType
      /**
       * 将某项更改设为选中状态，并将光标置于该更改范围的起始位置。
       *
       * @param id 要选中的更改 ID
       */
      selectAiChange: (id: string) => ReturnType
    }
  }
}
```

## 扩展存储

```ts
/**
 * AI Changes 扩展的内部存储属性。
 */
export interface AiChangesStorage {
  /** AI Changes 扩展使用的 ProseMirror 插件的键 */
  pluginKey: PluginKey<AiChangesProsemirrorPluginState>
  /**
   * 返回 AI 对文档所做的更改列表
   *
   * @returns 文档更改列表
   */
  getChanges: () => AiChange[]
  /**
   * 返回文档中当前选中的更改。光标位于所选更改的范围内。
   *
   * @returns 当前选中的文档更改；如果未选中任何更改，则返回 null
   */
  getSelectedChange: () => AiChange | null
  /**
   * 返回 AI 更改前的文档。如果扩展当前未跟踪更改，则返回 `null`。
   *
   * @returns AI 更改前的文档；如果未跟踪任何更改，则返回 null
   */
  getPreviousDoc: () => Node | null
  /**
   * 如果扩展当前正在跟踪 AI 所做的更改，则返回 `true`；否则返回 `false`。
   *
   * @returns 如果扩展当前正在跟踪 AI 所做的更改，则返回 `true`；否则返回 `false`
   */
  getIsTrackingAiChanges: () => boolean
  /**
   * 如果扩展当前正在显示 AI 所做的更改，则返回 `true`；否则返回 `false`。
   *
   * @returns 如果扩展当前正在显示 AI 所做的更改，则返回 `true`；否则返回 `false`
   */
  getIsShowingAiChanges: () => boolean
}
```

## 扩展类型

### 更改

```ts
import { Range } from '@tiptap/core'

/**
 * AI 代理所做的一项更改。通过比较新旧文档（即 AI 代理更改前后的文档）计算差异，
 * 得到更改对象。更改对象存储在 AI Changes 扩展的存储中，并用于在编辑器中呈现更改。
 */
export interface AiChange {
  /**
   * 更改的唯一标识符。
   */
  id: string
  /**
   * 旧文档中已删除内容的范围。
   */
  oldRange: Range
  /**
   * 新文档中新增内容的范围。
   * 这是 AI 代理插入内容的范围。
   */
  newRange: Range
}
```
